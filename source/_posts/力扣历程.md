### 两数之和

给定一个整数数组`nums`和一个整数目标值`target`，请你在该数组中找出 和为目标值`target`的那 两个 整数，并返回它们的数组下标。

你可以假设每种输入只会对应一个答案，并且你不能使用两次相同的元素。

你可以按任意顺序返回答案。

{% tabs 两数之和, 1 %}
<!-- tab C++ -->
```cpp
#include <vector>
#include <unordered_map>

using namespace std;

class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        // 创建哈希表，key为数组元素的值，value为元素的下标
        unordered_map<int, int> num_map;
        
        for (int i = 0; i < nums.size(); ++i) {
            int complement = target - nums[i];
            
            // 在哈希表中查找补数是否存在
            auto it = num_map.find(complement);
            if (it != num_map.end()) {
                // 找到则返回两个下标
                return {it->second, i};
            }
            
            // 未找到则将当前元素及下标存入哈希表
            num_map[nums[i]] = i;
        }
        
        return {}; // 题目保证有解，此处仅为语法要求
    }
};
```

import java.util.HashMap;
import java.util.Map;

class Solution {
    public int[] twoSum(int[] nums, int target) {
        // 创建 HashMap，key为数组元素的值，value为元素的下标
        Map<Integer, Integer> map = new HashMap<>();
        
        for (int i = 0; i < nums.length; i++) {
            int complement = target - nums[i];
            
            // 判断 HashMap 中是否包含目标差值
            if (map.containsKey(complement)) {
                // 找到则返回两个下标
                return new int[] { map.get(complement), i };
            }
            
            // 未找到则将当前元素及下标存入 HashMap
            map.put(nums[i], i);
        }
        
        // 题目保证有解，实际不会走到这里
        throw new IllegalArgumentException("No two sum solution");
    }
}

### 66.加一
数组 `digits` 从左到右保存一个非负整数的各位数字，例如 `[3, 0, 8]` 表示 308。 数字 0 用`[0]`表示；其他输入的第一位均不为 0。

请模拟一次十进制竖式加法，将这个整数增加 1，并把结果仍按相同的数组形式返回。 可以直接修改 `digits`；若最前面产生新的进位，可以返回一个更长的数组。

{% tabs 加一, 1 %}

<!-- tab C++ -->
```cpp
class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        std::vector<int> temp = digits;

        int length = temp.size();

        if (length == 1 && temp[0] < 9) {
            temp[0] += 1;
            return temp;
        }

        if (temp[length - 1] != 9) {
            temp[length - 1] += 1;
            return temp;
        }
        else {
            temp[length - 1] += 1;
            for (int i = length - 1; i > 0; i--) {
                if (temp[i] > 9) {
                    temp[i] = 0;
                    temp[i - 1] += 1;
                }
            }
            
        }

        if (temp[0] > 9 || (length == 1 && temp[0] >= 9)) {
            temp[0] = 0;
            temp.insert(temp.begin(), 1);
        }

        return temp;
    }
};
```

<!-- tab C++更优解 -->
```cpp
class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        int n = digits.size();
        
        // 从最后一位开始向前遍历
        for (int i = n - 1; i >= 0; i--) {
            // 如果当前位小于 9，直接加 1 并返回
            if (digits[i] < 9) {
                digits[i]++;
                return digits; 
            }
            // 如果当前位是 9，把它变成 0，然后继续循环处理前一位
            digits[i] = 0;
        }
        
        // 如果循环结束还没返回，说明数组里全是 9，例如 [9, 9, 9]
        // 此时所有位都变成了 0，只需要在头部插入一个 1 即可
        digits.insert(digits.begin(), 1);
        
        return digits;
    }
};
```

### 206.反转链表

给定一个单向链表的头节点 `head`。每个节点包含一个值和指向下一节点的 `next`。

请使用迭代方法反转所有节点之间的连接，并返回反转后的头节点。 要求不创建新的链表节点。

{% tabs 反转链表, 1 %}
<!-- tab C++ -->
```cpp
struct ListNode {
    int val;
    ListNode *next;


    ListNode() : val(0), next(nullptr) {}
    ListNode(int x) : val(x), next(nullptr) {}
    ListNode(int x, ListNode *next) : val(x), next(next) {}
};

class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode *cur = head, *prev = nullptr;

        while (cur) {
            ListNode* tmp = cur->next;
            cur->next = prev;
            prev = cur;
            cur = tmp;
        }

        return prev;
    }
};
```

### 20.有效的括号

给定一个只包含 `()`、`[]`、`{}` 这三类括号的字符串 `s`，请使用栈判断它是否合法。

合法序列须同时满足：每个右括号都必须与最近一个尚未配对的左括号类型匹配， 并且遍历结束后没有未配对的左括号。返回布尔值表示判断结果。

{% tabs 有效的括号, 1 %}
<!-- tab C++ -->
```cpp
#include <stack>
#include <string>
using namespace std;

class Solution {
public:
    bool isValid(string s) {
        stack<char> st;

        for (char c : s) {
            // 左括号入栈
            if (c == '(' || c == '[' || c == '{') {
                st.push(c);
            } 
            else {
                // 右括号，但栈为空，说明没有对应左括号
                if (st.empty()) return false;

                char top = st.top();

                // 检查是否匹配
                if ((c == ')' && top != '(') ||
                    (c == ']' && top != '[') ||
                    (c == '}' && top != '{')) {
                    return false;
                }

                // 匹配成功，弹出栈顶左括号
                st.pop();
            }
        }

        // 最后栈为空，说明所有左括号都被匹配
        return st.empty();
    }
};
```
